def analisar_licitacao_alimentacao(data):
    """
    Analisa dados de licitação para determinar se representa uma oportunidade na área de alimentação e nutrição.
    """

    razão_social = data.get('orgaoEntidade', {}).get('razaoSocial')
    numero_compra = data.get('numeroCompra')
    objeto_compra = data.get('objetoCompra')
    informacao_complementar = data.get('informacaoComplementar')
    tipo_instrumento_convocatorio_nome = data.get('tipoInstrumentoConvocatorioNome')

    if not objeto_compra or not informacao_complementar:
        return "Proposta " + numero_compra + " não é alimentação"

    if "material educativo e esportivo" in objeto_compra.lower() or "ano letivo" in objeto_compra.lower():
        return "Proposta " + numero_compra + " não é alimentação"

    if "aquisição de material educativo e esportivo" in informacao_complementar.lower():
        return "Proposta " + numero_compra + " não é alimentação"
